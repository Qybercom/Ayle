Ayle / Low-level API

Four canonical low-level examples

Each card is runnable and shows the exact JavaScript used below it: minimal video, minimal audio, full video and full audio.

Minimal UI + video media mode.

var root = document.getElementById('player-minimal-video');
var video = root.querySelector('.ayle-video');
var driver = new AyleMSEMediaDriver(video);
var player = new Ayle(driver, {
	AutoPlay: false,
	AutoFocus: true,
	MediaMode: 'video',
	UIMode: 'minimal',
	Volume: 0.8,
	Muted: false,
	Start: 0,
	MinimalUI: {
		Play: true,
		Timeline: true,
		Time: true,
		Volume: true,
		Settings: false,
		Chapters: false,
		Audio: false,
		Subtitles: false,
		Quality: false,
		PictureInPicture: false,
		Fullscreen: false,
		Header: false,
		Hints: false,
		Loading: true
	}
});
var ui = new AyleUI(root, player);

player.On('ready', function () {
	console.log('player-minimal-video ready');
});

player.On('error', function (error) {
	console.error('player-minimal-video error:', error);
});

var http = new AyleHTTP(player, {
	File: 'example.mkv',
	MetadataURL: '../../server/metadata.php?file={file}',
	TrackURL: '../../server/track.php?file={file}&type={kind}&track={track}&start={time}',
	Stream: {
		SkipInit: true
	}
});

http.Load(function (error, source, metadata) {
	if (error)
		console.error(error);
	else
		console.log(source, metadata);
});
