Ayle / Low-level API

Four canonical low-level examples

The page keeps the four manual low-level variants, the Ayle.Init() shortcut, and now also shows explicit programmatic AyleBootstrap initialization for video and audio.

Manual low-level assembly
var player = new Ayle({
	Driver: {
		Type: 'html5'
	},
	MediaProvider: {
		File: 'example.mp4'
	},
	Player: {
		MediaMode: 'video'
	}
});
player.AttachUI('#player-minimal-video');

player.On('ready', function () {
	console.log('player-minimal-video ready');
});

player.On('error', function (error) {
	console.error('player-minimal-video error:', error);
});

player.Load();
Same player with Ayle.Init()
var player = Ayle.Init('#player-minimal-video', {
	Driver: {
		Type: 'html5'
	},
	MediaProvider: {
		File: 'example.mp4'
	},
	Player: {
		MediaMode: 'video'
	}
});

AyleBootstrap video

Programmatic Bootstrap initialization using the built-in video preset.

bootstrap-video
Bootstrap config
<script
	type="application/json"
	data-ayle="bootstrap-video"
	data-ayle-preset="video"
	data-ayle-auto="false">
{
	"Driver": {
		"Type": "mse"
	},
	"MediaProvider": {
		"Type": "http",
		"File": "example.mkv",
		"MetadataURL": "../../server/metadata.php?file={file}",
		"TrackURL": "../../server/track.php?file={file}&type={kind}&track={track}&start={time}",
		"Stream": {
			"SkipInit": true
		}
	}
}
</script>
JavaScript
var bootstrap = new AyleBootstrap();
var config = document.querySelector('[data-ayle="bootstrap-video"]');

bootstrap.InitConfig(config);
